# जेमिनी का एकदम लेटेस्ट चालू मॉडल लोड करना
model = genai.GenerativeModel(
    model_name="gemini-3.8-flash",
    system_instruction=system_instruction
)

@bot.message_handler(func=lambda message: True)
def reply_to_chat(message):
    user_text = message.text
    chat_id = message.chat.id
    
    # ग्रुप चैट में जब कोई रितु को रिप्लाई करेगा या नाम लेगा, तभी वह बोलेगी (पर्सनल चैट में हमेशा बोलेगी)
    bot_username = bot.get_me().username
    if message.chat.type != "private":
        if f"@{bot_username}" not in user_text and (message.reply_to_message is None or message.reply_to_message.from_user.id != bot.get_me().id):
            return

    try:
        response = model.generate_content(user_text)
        bot_reply = response.text
        bot.reply_to(message, bot_reply)
    except Exception as e:
        print(f"Error: {e}")
